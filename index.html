// --- 核心：透過 CORS 代理伺服器抓取台股 OpenAPI 報價 --- async function fetchTaiwanStockPrices() { const btn = document.getElementById('api-update-btn'); const originalText = btn.innerHTML; btn.innerHTML = '⏳ 正在透過代理伺服器抓取資料...'; btn.style.opacity = '0.7'; btn.disabled = true; try { const priceMap = {}; // 1. 抓取上市資料 (TWSE) - 加上 CORS 代理 const twseUrl = encodeURIComponent('https://openapi.twse.com.tw/v1/exchangeReport/STOCK_DAY_ALL'); const twseRes = await fetch(`https://api.allorigins.win/raw?url=${twseUrl}`); if (twseRes.ok) { const twseData = await twseRes.json(); twseData.forEach(item => { const price = parseFloat(item.ClosingPrice); if (!isNaN(price)) { priceMap[item.Code] = price; priceMap[item.Name] = price; } }); } // 2. 抓取上櫃資料 (TPEx) - 加上 CORS 代理 const tpexUrl = encodeURIComponent('https://www.tpex.org.tw/openapi/v1/t187ap03_L'); const tpexRes = await fetch(`https://api.allorigins.win/raw?url=${tpexUrl}`); if (tpexRes.ok) { const tpexData = await tpexRes.json(); tpexData.forEach(item => { const price = parseFloat(item.Close); if (!isNaN(price)) { priceMap[item.SecuritiesCompanyCode] = price; priceMap[item.CompanyName] = price; } }); } // 3. 更新表格中的資料 let updatedCount = 0; targets.forEach((target, index) => { if (target.market === '台股' && target.name.trim() !== '') { const queryKey = target.name.trim(); const newPrice = priceMap[queryKey]; if (newPrice !== undefined && target.price !== newPrice) { target.price = newPrice; updatedCount++; setTimeout(() => { const inputEl = document.getElementById(`price-${index}`); if(inputEl) { inputEl.style.backgroundColor = '#e6f4ea'; setTimeout(() => inputEl.style.backgroundColor = 'transparent', 2000); } }, 100); } } }); if (updatedCount > 0) { saveTargets(); renderTargets(); alert(`✅ 成功為您更新了 ${updatedCount} 筆台股最新收盤價！`); } else { alert('⚠️ 資料抓取成功，但沒有找到對應的標的。請確認「市場」是否選擇台股，且名稱或代號是否正確填寫。'); } } catch (error) { console.error("API 抓取失敗:", error); alert('❌ 無法取得報價，可能是代理伺服器網路異常或證交所維護中。'); } finally { btn.innerHTML = originalText; btn.style.opacity = '1'; btn.disabled = false; } }
